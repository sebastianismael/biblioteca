private const val VIP = "VIP"
private const val BOOKS = "libros"
private const val MINIMAL_AMOUNT_TO_FREE_SHIPMENT = 5

@Component
class CalculateShimpmentCost {

    operator fun invoke(client: String, products: Map<String, Int>) =
        if (client.isVip() && products.enoughBooks() && products.containsJustBooks()) 0 else 10
}

private fun Map<String, Int>.containsJustBooks()=
    this.size == 1 && this.containsKey("BOOK")

private fun Map<String, Int>.enoughBooks() = this["BOOK"]?.let { books -> books >= 5 } ?: false

private fun String.isVip() = this == "VIP"
